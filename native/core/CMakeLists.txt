cmake_minimum_required(VERSION 3.22)
project(dae_pcm_core LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_library(dae_pcm STATIC dae_pcm.c)
set_target_properties(dae_pcm PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(dae_pcm PUBLIC include)
target_link_libraries(dae_pcm PUBLIC Threads::Threads m)
target_compile_options(dae_pcm PRIVATE -Wall -Wextra -Wpedantic -Werror)
option(DAE_BUILD_TESTS "Build native PCM tests" ON)
if(DAE_BUILD_TESTS)
  enable_testing()
  add_executable(dae_pcm_tests tests/pcm_test.c)
  target_link_libraries(dae_pcm_tests PRIVATE dae_pcm)
  target_compile_options(dae_pcm_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME pcm_contract COMMAND dae_pcm_tests)
endif()
