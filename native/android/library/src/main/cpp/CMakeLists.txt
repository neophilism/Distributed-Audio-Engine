add_library(dae_audio_jni SHARED native_pcm_jni.c)
target_link_libraries(dae_audio_jni PRIVATE dae_pcm)
target_compile_options(dae_audio_jni PRIVATE -Wall -Wextra -Wpedantic -Werror)
